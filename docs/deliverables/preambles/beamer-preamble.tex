% Small, muted reference for the end of a bullet, so the claim leads and the
% citation recedes. Use only for trailing parenthetical references, e.g.
%   \item Literature is unclear on these programs \smallcitation{(Smith 2008; Jones 2010)}
% Not for load-bearing inline citations, e.g. "the Rozendaal (2008) procedure".
% Styles its argument only; supply the parentheses or \citep{...} yourself.
\newcommand{\smallcitation}[1]{{\footnotesize\color{slidemuted}#1}}

\newenvironment{transitionframe}{%
  \begingroup
  \setbeamercolor{background canvas}{bg=slidetransitionbg}%
  \setbeamertemplate{footline}{}%
  \begin{frame}[plain,noframenumbering]
}{%
  \end{frame}
  \endgroup
}

\newcommand{\transitiontitle}[2][]{%
  \vspace*{0.19\paperheight}%
  \hspace*{0.055\paperwidth}%
  \begin{minipage}{0.78\paperwidth}
    {\color{slidegreen}\rule{0.17\paperwidth}{1.2pt}}\par
    \vspace{1.2em}
    {\color{slideink}\fontsize{25}{30}\selectfont\bfseries #2\par}
    \if\relax\detokenize{#1}\relax
    \else
      \vspace{0.7em}
      {\color{slidemuted}\fontsize{13}{17}\selectfont #1\par}
    \fi
  \end{minipage}%
  \vfill
}

\newcommand{\sectiontransition}[2][]{%
  \begin{transitionframe}
    \transitiontitle[#1]{#2}
  \end{transitionframe}
}
